## Contrato de mapeo (función ↔ posición ↔ pin Blend Micro ↔ GPIO ↔ ref)

| Ref | Función | Pos (x,y) | Footprint | Header.pin | GPIO |
|---|---|---|---|---|---|
| SW1 | Arriba | (32,32) | Redragon | J1.12 | D3 |
| SW2 | Abajo | (32,48) | Redragon | J1.10 | D1 |
| SW3 | Izquierda | (16,48) | Redragon | J1.11 | D2 |
| SW4 | Derecha | (48,48) | Redragon | J1.9 | D0 |
| SW5 | A | (64,32) | Redragon | J2.12 | D11 |
| SW6 | B | (80,32) | Redragon | J2.11 | D12 |
| SW7 | X | (64,48) | Redragon | J2.9 | A0 |
| SW8 | Y | (80,48) | Redragon | J2.8 | A1 |
| SW9 | L | (16,16) | Redragon | J1.14 | D8 |
| SW10 | R | (80,16) | Redragon | J2.14 | D9 |
| SW11 | Select | (32,16) | **tact CAX** | J1.13 | D5 |
| SW12 | Start | (64,16) | **tact CAX** | J2.13 | D10 |

GND: pad 2 de cada switch + J1.1, J1.2, J2.1. **Sin** nets SDA/SCL/+3V3 ni J3 (OLED fuera).
Pines evitados: D6/D7 (nRF8001, no en header), D13 (J2.10, LED), SPI (J1.4–8), AREF (J2.3), VIN (J1.3).

### Task 5: Firmware XInput a 12 botones

**Files:**
- Modify: `firmware/redbearpad_xinput/redbearpad_xinput.ino`

- [ ] **Step 1: Reemplazar el mapeo y la lógica** del sketch por la versión de 12 entradas (pines según el contrato; orden del array por función):

```cpp
#include <XInput.h>

// PIN_BOTON en orden de funcion. Pines segun el contrato del PCB v2.
// 0:Up(D3) 1:Down(D1) 2:Left(D2) 3:Right(D0) 4:A(D11) 5:B(D12) 6:X(A0) 7:Y(A1)
// 8:L(D8) 9:R(D9) 10:Select(D5) 11:Start(D10)
const uint8_t NUM_BOTONES = 12;
const uint8_t PIN_BOTON[NUM_BOTONES] = { 3, 1, 2, 0, 11, 12, A0, A1, 8, 9, 5, 10 };
const uint8_t IDX_UP = 0, IDX_DOWN = 1, IDX_LEFT = 2, IDX_RIGHT = 3;

bool estadoPrev[NUM_BOTONES];
uint8_t integrador[NUM_BOTONES];
uint32_t tUltimoSample = 0;
const uint8_t SAMPLE_MS = 1;
const uint8_t INTEGRADOR_MAX = 12;

void setup() {
  for (uint8_t i = 0; i < NUM_BOTONES; i++) {
    pinMode(PIN_BOTON[i], INPUT_PULLUP);
    estadoPrev[i] = false;
    integrador[i] = 0;
  }
  pinMode(LED_BUILTIN, OUTPUT);
  XInput.setAutoSend(false);
  XInput.begin();
}

void loop() {
  uint32_t ahora = millis();
  bool huboCambio = false;

  if (ahora - tUltimoSample >= SAMPLE_MS) {
    tUltimoSample = ahora;
    for (uint8_t i = 0; i < NUM_BOTONES; i++) {
      bool leido = (digitalRead(PIN_BOTON[i]) == LOW);
      if (leido) { if (integrador[i] < INTEGRADOR_MAX) integrador[i]++; }
      else       { if (integrador[i] > 0)              integrador[i]--; }
      bool nuevo = estadoPrev[i];
      if (integrador[i] >= INTEGRADOR_MAX) nuevo = true;
      else if (integrador[i] == 0)         nuevo = false;
      if (nuevo != estadoPrev[i]) { estadoPrev[i] = nuevo; huboCambio = true; }
    }
  }

  if (huboCambio) {
    bool up = estadoPrev[IDX_UP], down = estadoPrev[IDX_DOWN];
    bool left = estadoPrev[IDX_LEFT], right = estadoPrev[IDX_RIGHT];
    XInput.setJoystick(JOY_LEFT, up, down, left, right);
    XInput.setDpad(up, down, left, right);
    XInput.setButton(BUTTON_A,     estadoPrev[4]);
    XInput.setButton(BUTTON_B,     estadoPrev[5]);
    XInput.setButton(BUTTON_X,     estadoPrev[6]);
    XInput.setButton(BUTTON_Y,     estadoPrev[7]);
    XInput.setButton(BUTTON_LB,    estadoPrev[8]);   // L
    XInput.setButton(BUTTON_RB,    estadoPrev[9]);   // R
    XInput.setButton(BUTTON_BACK,  estadoPrev[10]);  // Select
    XInput.setButton(BUTTON_START, estadoPrev[11]);  // Start
    XInput.send();
  }

  digitalWrite(LED_BUILTIN, (millis() / 500) % 2);
}
```
